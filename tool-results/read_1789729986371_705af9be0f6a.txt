    56→  const setOpenForm = useHrmsStore((s) => s.setOpenForm);
    57→  const queryClient = useQueryClient();

    61→
    62→  const leaveQuery = useQuery({
    63→    queryKey: ["leave"],
    64→    queryFn: () => apiGet<LeaveData>("/api/leave"),
    65→  });
    66→
    67→  // auto-open apply dialog when navigated with a form trigger (quick create / desk CTA)

   326→// ── apply leave dialog ──────────────────────────────────────
   327→function ApplyLeaveDialog({
   328→  open, onOpenChange, balances, onApplied,
   329→}: {
   330→  open: boolean;
   331→  onOpenChange: (v: boolean) => void;
   332→  balances: LeaveBalance[];
   333→  onApplied: () => void;
   334→}) {
   335→  const today = useMemo(() => isoDay(dayIST(new Date())), []);
   336→  const [form, setForm] = useState({
   337→    leaveTypeId: "", fromDate: "", toDate: "", dayMode: "FULL" as DayMode,
   338→    reason: "", contact: "", substituteId: "none", attachmentName: "",
   339→  });
   340→  const [errors, setErrors] = useState<{ leaveTypeId?: string; dates?: string; reason?: string; balance?: string }>({});
   341→
   342→  const subsQuery = useQuery({
   343→    queryKey: ["leave", "substitutes"],
   344→    queryFn: () => apiGet<{ substitutes: Substitute[] }>("/api/leave?action=substitutes"),
   345→    enabled: open,
   346→  });
   347→
   348→  const selectedBalance = balances.find((b) => b.leaveTypeId === form.leaveTypeId) ?? null;
   349→
   350→  const rangeDays = useMemo(() => {
   351→    if (!form.fromDate || !form.toDate) return 0;
   352→    const from = new Date(`${form.fromDate}T00:00:00Z`).getTime();
   353→    const to = new Date(`${form.toDate}T00:00:00Z`).getTime();
   354→    if (Number.isNaN(from) || Number.isNaN(to) || from > to) return 0;
   355→    return Math.round((to - from) / 86400000) + 1;
   356→  }, [form.fromDate, form.toDate]);
   357→  const days = rangeDays * (form.dayMode === "FULL" ? 1 : 0.5);
   358→  const balanceShort = selectedBalance ? days > selectedBalance.available : false;
   359→
   360→  function reset() {
   361→    setForm({ leaveTypeId: "", fromDate: "", toDate: "", dayMode: "FULL", reason: "", contact: "", substituteId: "none", attachmentName: "" });
   362→    setErrors({});
   363→  }
   364→
   365→  const mutation = useMutation({
   366→    mutationFn: () =>
   367→      apiPost<{ code: string; days: number }>("/api/leave", {
   368→        action: "create",
   369→        leaveTypeId: form.leaveTypeId,
   370→        fromDate: form.fromDate,
   371→        toDate: form.toDate || form.fromDate,
   372→        dayMode: form.dayMode,
   373→        reason: form.reason.trim(),
   374→        contactDuringLeave: form.contact.trim() || undefined,
   375→        substituteId: form.substituteId === "none" ? undefined : form.substituteId,
   376→        attachmentName: form.attachmentName.trim() || undefined,
   377→      }),
   378→    onSuccess: (res) => {
   379→      toast.success(`Leave request ${res.code} submitted`, {
   380→        description: `${res.days} day(s) blocked · awaiting manager approval.`,
   381→      });
   382→      onOpenChange(false);
   383→      reset();
   384→      onApplied();
   385→    },
   386→    onError: (err) => {
   387→      toast.error(err instanceof ApiError ? err.message : "Could not submit the request.");
   388→    },
   389→  });
   390→
   391→  function submit() {
   392→    const errs: typeof errors = {};
   393→    if (!form.leaveTypeId) errs.leaveTypeId = "Select a leave type.";
   394→    if (!form.fromDate || !form.toDate) errs.dates = "Select both dates.";
   395→    else if (form.fromDate > form.toDate) errs.dates = "From date cannot be after to date.";
   396→    if (form.reason.trim().length < 5) errs.reason = "Reason must be at least 5 characters.";
   397→    if (selectedBalance && days > selectedBalance.available) {
   398→      errs.balance = `Only ${selectedBalance.available} day(s) available for ${selectedBalance.name}.`;
   399→    }
   400→    setErrors(errs);
   401→    if (Object.keys(errs).length > 0) return;
   402→    mutation.mutate();
   403→  }
   404→
   405→  return (
   406→    <Dialog open={open} onOpenChange={(v) => { onOpenChange(v); if (!v) reset(); }}>
   407→      <DialogContent className="max-h-[90vh] overflow-y-auto scroll-thin sm:max-w-lg">
   408→        <DialogHeader>
   409→          <DialogTitle className="flex items-center gap-2">
   410→            <CalendarPlus className="h-4 w-4 text-primary" /> Apply for Leave
   411→          </DialogTitle>
   412→          <DialogDescription>Submit a leave request for manager and HR approval.</DialogDescription>
   413→        </DialogHeader>
   414→
   415→        <div className="space-y-4">
   416→          <div className="space-y-1.5">
   417→            <Label>Leave Type</Label>
   418→            <Select value={form.leaveTypeId} onValueChange={(v) => setForm((f) => ({ ...f, leaveTypeId: v }))}>
   419→              <SelectTrigger className="text-sm">
   420→                <SelectValue placeholder="Select leave type" />
   421→              </SelectTrigger>
   422→              <SelectContent>
   423→                {balances.map((b) => (
   424→                  <SelectItem key={b.leaveTypeId} value={b.leaveTypeId}>
   425→                    <span className="flex items-center gap-2">
   426→                      <span className="h-2 w-2 rounded-full" style={{ backgroundColor: b.color }} aria-hidden />
   427→                      {b.name}
   428→                      <span className="text-[10px] text-muted-foreground">{b.available}d available</span>
   429→                    </span>
   430→                  </SelectItem>
   431→                ))}
   432→              </SelectContent>
   433→            </Select>
   434→            {errors.leaveTypeId ? <p className="text-xs text-danger">{errors.leaveTypeId}</p> : null}
   435→            {selectedBalance ? (
   436→              <p className="text-[11px] text-muted-foreground">
   437→                {selectedBalance.available} of {selectedBalance.entitled} day(s) available · {selectedBalance.pending} pending
   438→              </p>
   439→            ) : null}
   440→          </div>
   441→
   442→          <div className="grid grid-cols-2 gap-3">
   443→            <div className="space-y-1.5">
   444→              <Label htmlFor="lv-from">From</Label>
   445→              <Input
   446→                id="lv-from" type="date" value={form.fromDate}
   447→                onChange={(e) => setForm((f) => ({
   448→                  ...f, fromDate: e.target.value,
   449→                  toDate: !f.toDate || f.toDate < e.target.value ? e.target.value : f.toDate,
   450→                }))}
   451→              />
   452→            </div>
   453→            <div className="space-y-1.5">
   454→              <Label htmlFor="lv-to">To</Label>
   455→              <Input
   456→                id="lv-to" type="date" min={form.fromDate || today} value={form.toDate}
   457→                onChange={(e) => setForm((f) => ({ ...f, toDate: e.target.value }))}
   458→              />
   459→            </div>
   460→          </div>
   461→          {errors.dates ? <p className="text-xs text-danger">{errors.dates}</p> : null}
   462→
   463→          <div className="space-y-1.5">
   464→            <Label>Day Mode</Label>
   465→            <RadioGroup
   466→              value={form.dayMode}
   467→              onValueChange={(v) => setForm((f) => ({ ...f, dayMode: v as DayMode }))}
   468→              className="grid grid-cols-3 gap-2"
   469→            >
