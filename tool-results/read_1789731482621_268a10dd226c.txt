    34→  reason: string; contactDuringLeave: string | null; substituteName: string | null;
    35→  appliedAt: string; stage: "MANAGER" | "HR";

    42→  clientName: string | null; address: string | null; appliedAt: string;
    43→}

    49→  reason: string; appliedAt: string;
    50→}
    51→
    52→interface TimesheetGroup {
    53→  weekStart: string; employee: EmployeeBrief;

    57→
    58→interface ExpenseItem {
    59→  id: string; code: string; employee: EmployeeBrief; category: string;
    60→  expenseDate: string; amount: number; currency: string; merchant: string | null;
    61→  project: string | null; description: string | null; status: string;
    62→  submittedAt: string | null; stage: "MANAGER" | "FINANCE" | "PAY";
    63→}

   110→  disabled?: boolean;

   118→          variant="outline"
   119→          className="h-8 gap-1.5 border-success/40 text-xs text-success hover:bg-success-soft"

   137→        <Button
   138→          size="sm"
   139→          variant="outline"
   140→          className="h-8 gap-1.5 border-danger/40 text-xs text-danger hover:bg-danger-soft"

   151→export default function ApprovalsView() {
   152→  const employee = useHrmsStore((s) => s.employee);
   153→  const timeFormat = useHrmsStore((s) => s.timeFormat);
   154→  const queryClient = useQueryClient();
   155→  const isManager = employee?.role === "MANAGER" || employee?.role === "HR" || employee?.role === "ADMIN";
   156→
   157→  const [tab, setTab] = useState<TabKey>("leave");
   158→  const [draft, setDraft] = useState<DecisionDraft | null>(null);
   159→  const [comment, setComment] = useState("");
   160→
   161→  const approvalsQuery = useQuery({
   162→    queryKey: ["approvals", tab],
   163→    queryFn: () => apiGet<ApprovalsResponse>(`/api/approvals?tab=${tab}`),
   164→    enabled: isManager,
   165→  });
   166→
   167→  const decideMutation = useMutation({
   168→    mutationFn: (vars: { kind: DecidableKind; ids: string[]; action: DecidableAction; comment?: string }) =>
   169→      apiPost<{ processed: number }>("/api/approvals/decide", {
   170→        kind: vars.kind,
   171→        ids: vars.ids,
   172→        action: vars.action,
   173→        ...(vars.comment ? { comment: vars.comment } : {}),
   174→      }),
   175→    onSuccess: (res, vars) => {
   176→      const verb = vars.action === "approve" ? "approved" : vars.action === "pay" ? "marked paid" : "rejected";
   177→      toast.success(`${res.processed} item${res.processed === 1 ? "" : "s"} ${verb}`);
   178→      setDraft(null);
   179→      setComment("");
   180→      void queryClient.invalidateQueries({ queryKey: ["approvals"] });
   181→      void queryClient.invalidateQueries({ queryKey: ["desk"] });
   182→    },
   183→    onError: (err) => {
   184→      toast.error(err instanceof ApiError ? err.message : "Decision could not be recorded");
   185→    },
   186→  });
   187→
   188→  const counts = approvalsQuery.data?.counts;
   189→
   190→  const items = useMemo(() => approvalsQuery.data?.items ?? [], [approvalsQuery.data]);
   191→
   192→  function openDecision(kind: DecidableKind, action: DecidableAction, ids: string[], employeeName: string, code: string) {
   193→    setComment("");
   194→    setDraft({ kind, action, ids, employeeName, code });
   195→  }
   196→
   197→  function confirmDecision() {
   198→    if (!draft) return;
   199→    decideMutation.mutate({ kind: draft.kind, ids: draft.ids, action: draft.action, comment: comment.trim() || undefined });
   200→  }
   201→
   202→  if (!isManager) {
   203→    return (
   204→      <div className="space-y-4">
   205→        <PageHeader title="Approvals" subtitle="Pending requests from your team" icon={<ClipboardCheck className="h-4.5 w-4.5" />} />
   206→        <EmptyState
   207→          title="Approvals requires a manager role"
   208→          message="Only managers, HR and admins can review team requests."
   209→          icon={<UserX className="h-5 w-5" />}
   210→        />
   211→      </div>
   212→    );
   213→  }
   214→
   215→  return (
   216→    <div className="space-y-4">
   217→      <PageHeader
   218→        title="Approvals"
   219→        subtitle="Pending requests from your team"
   220→        icon={<ClipboardCheck className="h-4.5 w-4.5" />}
   221→        actions={
   222→          counts ? (
   223→            <Badge variant="outline" className="h-7 gap-1.5 bg-warning-soft px-2.5 text-xs text-[#B54708]">
   224→              <Clock3 className="h-3.5 w-3.5" />
   225→              {counts.total} pending
   226→            </Badge>
   227→          ) : null
   228→        }
   229→      />
   230→
   231→      <Tabs value={tab} onValueChange={(v) => setTab(v as TabKey)}>
   232→        <div className="w-full overflow-x-auto pb-1">
   233→          <TabsList className="h-9 w-max">
   234→            {(Object.keys(TAB_LABELS) as TabKey[]).map((key) => (
   235→              <TabsTrigger key={key} value={key} className="gap-1.5 text-xs">
   236→                {TAB_LABELS[key]}
   237→                {counts ? (
   238→                  <Badge
   239→                    variant="secondary"
   240→                    className={`ml-0.5 h-4 min-w-4 px-1 text-[10px] tabular ${counts[key] > 0 ? "bg-primary/10 text-primary" : "text-muted-foreground"}`}
   241→                  >
   242→                    {counts[key]}
   243→                  </Badge>
   244→                ) : null}
   245→              </TabsTrigger>
   246→            ))}
   247→          </TabsList>
   248→        </div>
   249→      </Tabs>
   250→
   251→      <DataState query={approvalsQuery}>
   252→        {() => (
   253→          <div className="max-h-[62vh] space-y-3 overflow-y-auto scroll-thin pr-1">
   254→            {items.length === 0 ? (
   255→              <EmptyState
   256→                title="No pending approvals. Inbox zero!"
   257→                message={`All ${TAB_LABELS[tab].toLowerCase()} requests from your team are handled.`}
   258→                icon={<CheckCheck className="h-5 w-5" />}
   259→              />
   260→            ) : (
   261→              items.map((item, index) => {
   262→                const busy = decideMutation.isPending;
   263→                if (tab === "leave") {
   264→                  const l = item as LeaveItem;
   265→                  return (
   266→                    <LeaveRow
   267→                      key={l.id}
   268→                      item={l}
   269→                      disabled={busy}
   270→                      onApprove={() => openDecision("leave", "approve", [l.id], l.employee.name, l.code)}
   271→                      onReject={() => openDecision("leave", "reject", [l.id], l.employee.name, l.code)}
   272→                    />
   273→                  );
   274→                }
   275→                if (tab === "duty") {
   276→                  const d = item as DutyItem;
   277→                  return (
   278→                    <DutyRow
   279→                      key={d.id}
   280→                      item={d}
   281→                      disabled={busy}
   282→                      onApprove={() => openDecision("duty", "approve", [d.id], d.employee.name, d.code)}
   283→                      onReject={() => openDecision("duty", "reject", [d.id], d.employee.name, d.code)}
   284→                    />
   285→                  );
   286→                }
   287→                if (tab === "attendance") {
   288→                  const a = item as AttendanceItem;
   289→                  return (
   290→                    <AttendanceRow
   291→                      key={a.id}
   292→                      item={a}
   293→                      timeFormat={timeFormat}
   294→                      disabled={busy}
   295→                      onApprove={() => openDecision("attendance", "approve", [a.id], a.employee.name, a.code)}
   296→                      onReject={() => openDecision("attendance", "reject", [a.id], a.employee.name, a.code)}
   297→                    />
   298→                  );
   299→                }
   300→                if (tab === "timesheet") {
   301→                  const g = item as TimesheetGroup;
   302→                  return (
   303→                    <TimesheetRow
   304→                      key={`${g.weekStart}-${g.employee.id}-${index}`}
   305→                      item={g}
   306→                      disabled={busy}
   307→                      onApprove={() => openDecision("timesheet", "approve", g.entries.map((e) => e.id), g.employee.name, `week of ${fmtDateShort(g.weekStart)}`)}
   308→                      onReject={() => openDecision("timesheet", "reject", g.entries.map((e) => e.id), g.employee.name, `week of ${fmtDateShort(g.weekStart)}`)}
   309→                    />
   310→                  );
   311→                }
   312→                const x = item as ExpenseItem;
   313→                return (
   314→                  <ExpenseRow
   315→                    key={x.id}
   316→                    item={x}
   317→                    disabled={busy}
   318→                    onApprove={() => openDecision("expense", "approve", [x.id], x.employee.name, x.code)}
   319→                    onReject={x.stage !== "PAY" ? () => openDecision("expense", "reject", [x.id], x.employee.name, x.code) : undefined}
   320→                    onPay={() => openDecision("expense", "pay", [x.id], x.employee.name, x.code)}
   321→                  />
   322→                );
   323→              })
   324→            )}
   325→          </div>
   326→        )}
   327→      </DataState>
   328→
   329→      {/* decision dialog */}
   330→      <Dialog open={draft !== null} onOpenChange={(open) => { if (!open && !decideMutation.isPending) setDraft(null); }}>
   331→        <DialogContent className="sm:max-w-md">
   332→          {draft ? (
   333→            <>
   334→              <DialogHeader>
   335→                <DialogTitle>
   336→                  {draft.action === "approve" ? "Approve" : draft.action === "pay" ? "Mark as paid" : "Reject"}{" "}
   337→                  {KIND_LABELS[draft.kind]}?
   338→                </DialogTitle>
   339→                <DialogDescription>
   340→                  {draft.employeeName} · {draft.code}
   341→                  {draft.action === "approve" && draft.kind === "leave" && draft.ids.length === 1
   342→                    ? " — approving forwards the request to HR for final confirmation."
   343→                    : ""}
   344→                </DialogDescription>
   345→              </DialogHeader>
   346→              <div className="space-y-2">
   347→                <Textarea
   348→                  value={comment}
   349→                  onChange={(e) => setComment(e.target.value)}
   350→                  placeholder="Optional comment for the requester…"
   351→                  className="min-h-20 text-sm"
   352→                  maxLength={500}
   353→                />
   354→              </div>
   355→              <DialogFooter className="gap-2">
   356→                <Button
   357→                  variant="ghost"
   358→                  size="sm"
   359→                  className="h-8 text-xs"
   360→                  onClick={() => setDraft(null)}
   361→                  disabled={decideMutation.isPending}
   362→                >
   363→                  Cancel
   364→                </Button>
   365→                <Button
   366→                  size="sm"
   367→                  className="h-8 gap-1.5 text-xs"
   368→                  variant={draft.action === "reject" ? "destructive" : "default"}
   369→                  onClick={confirmDecision}
   370→                  disabled={decideMutation.isPending}
   371→                >
   372→                  {decideMutation.isPending ? (
   373→                    <>
   374→                      <Loader2 className="h-3.5 w-3.5 animate-spin" /> Processing…
   375→                    </>
   376→                  ) : draft.action === "reject" ? (
   377→                    <>
   378→                      <ThumbsDown className="h-3.5 w-3.5" /> Confirm reject
   379→                    </>
   380→                  ) : draft.action === "pay" ? (
   381→                    <>
   382→                      <Banknote className="h-3.5 w-3.5" /> Confirm payment
   383→                    </>
   384→                  ) : (
   385→                    <>
   386→                      <ThumbsUp className="h-3.5 w-3.5" /> Confirm approval
   387→                    </>
   388→                  )}
   389→                </Button>
   390→              </DialogFooter>
   391→            </>
   392→          ) : null}
   393→        </DialogContent>
   394→      </Dialog>
   395→    </div>
   396→  );
   397→}

   401→function RowShell({ employee, code, topBadge, children, actions }: {
   402→  employee: EmployeeBrief;
   403→  code: string;
   404→  topBadge?: React.ReactNode;
   405→  children: React.ReactNode;
   406→  actions: React.ReactNode;
   407→}) {
   408→  const [firstName, ...rest] = employee.name.split(" ");
   409→  return (
   410→    <div className="rounded-xl border border-border bg-card p-3.5 shadow-none">
   411→      <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   412→        <div className="flex min-w-0 items-start gap-2.5">
   413→          <Initials first={firstName} last={rest.join(" ") || " "} size="sm" />
   414→          <div className="min-w-0">
   415→            <div className="flex flex-wrap items-center gap-1.5">
   416→              <p className="truncate text-sm font-semibold text-foreground">{employee.name}</p>
   417→              <span className="font-mono text-[10px] text-muted-foreground">{code}</span>
   418→              {topBadge}
   419→            </div>
   420→            <p className="truncate text-[11px] text-muted-foreground">{employee.designation}</p>
   421→          </div>
   422→        </div>
   423→        {actions}
   424→      </div>
   425→      <div className="mt-3 border-t border-border pt-2.5">{children}</div>
   426→    </div>
   427→  );
   428→}

   430→function LeaveRow({ item, disabled, onApprove, onReject }: {
   431→  item: LeaveItem;
   432→  disabled?: boolean;
   433→  onApprove: () => void;
   434→  onReject: () => void;
   435→}) {
   436→  return (
   437→    <RowShell
   438→      employee={item.employee}
   439→      code={item.code}
   440→      topBadge={
   441→        item.stage === "HR" ? (
   442→          <Badge variant="outline" className="bg-info-soft text-[10px] text-info">HR review</Badge>
   443→        ) : null
   444→      }
   445→      actions={<ActionButtons onApprove={onApprove} onReject={onReject} disabled={disabled} />}
   446→    >
   447→      <div className="grid gap-x-4 gap-y-1.5 text-xs sm:grid-cols-2">
   448→        <span className="flex items-center gap-1.5 text-muted-foreground">
   449→          <CalendarRange className="h-3.5 w-3.5" />
   450→          <span className="font-medium text-foreground">{fmtDateShort(item.fromDate)}</span>
   451→          {item.toDate !== item.fromDate ? (
   452→            <>
   453→              <ArrowRight className="h-3 w-3" />
   454→              <span className="font-medium text-foreground">{fmtDateShort(item.toDate)}</span>
   455→            </>
   456→          ) : null}
   457→          <span>· {item.days} day{item.days === 1 ? "" : "s"}</span>
   458→        </span>
   459→        <span className="flex items-center gap-1.5 text-muted-foreground">
   460→          <Badge
   461→            variant="outline"
   462→            className="px-1.5 py-0 text-[10px]"
   463→            style={{ color: item.leaveTypeColor, borderColor: `${item.leaveTypeColor}55` }}
   464→          >
   465→            {item.leaveType}
   466→          </Badge>
   467→          <span>{DAY_MODE_LABELS[item.dayMode] ?? item.dayMode}</span>
   468→        </span>
   469→        <span className="flex items-start gap-1.5 text-muted-foreground sm:col-span-2">
   470→          <FileText className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   471→          <span className="text-foreground">{item.reason}</span>
   472→        </span>
   473→        {item.contactDuringLeave || item.substituteName ? (
   474→          <span className="text-[11px] text-muted-foreground sm:col-span-2">
   475→            {item.contactDuringLeave ? `Contact: ${item.contactDuringLeave}` : ""}
   476→            {item.contactDuringLeave && item.substituteName ? " · " : ""}
   477→            {item.substituteName ? `Substitute: ${item.substituteName}` : ""}
   478→          </span>
   479→        ) : null}
   480→      </div>
   481→      <p className="mt-1.5 text-[10px] text-muted-foreground">Applied {fmtDate(item.appliedAt)}</p>
   482→    </RowShell>
   483→  );

   486→function DutyRow({ item, disabled, onApprove, onReject }: {
   487→  item: DutyItem;
   488→  disabled?: boolean;
   489→  onApprove: () => void;
   490→  onReject: () => void;
   491→}) {
   492→  return (
   493→    <RowShell
   494→      employee={item.employee}
   495→      code={item.code}
   496→      topBadge={<StatusBadge status={item.type} label={item.type === "WFH" ? "WFH" : "On Duty"} />}
   497→      actions={<ActionButtons onApprove={onApprove} onReject={onReject} disabled={disabled} />}
   498→    >
   499→      <div className="grid gap-x-4 gap-y-1.5 text-xs sm:grid-cols-2">
   500→        <span className="flex items-center gap-1.5 text-muted-foreground">
   501→          <CalendarRange className="h-3.5 w-3.5" />
   502→          <span className="font-medium text-foreground">{fmtDateShort(item.fromDate)}</span>
   503→          {item.toDate && item.toDate !== item.fromDate ? (
   504→            <>
   505→              <ArrowRight className="h-3 w-3" />
   506→              <span className="font-medium text-foreground">{fmtDateShort(item.toDate)}</span>
   507→            </>
   508→          ) : null}
   509→          {item.hours ? <span>· {item.hours}h</span> : null}
   510→        </span>
   511→        {item.subtype ? (
   512→          <span className="text-muted-foreground">
   513→            Type <span className="font-medium text-foreground">{item.subtype.replace(/_/g, " ")}</span>
   514→          </span>
   515→        ) : null}
   516→        <span className="flex items-start gap-1.5 text-muted-foreground sm:col-span-2">
   517→          <FileText className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   518→          <span className="text-foreground">{item.reason}</span>
   519→        </span>
