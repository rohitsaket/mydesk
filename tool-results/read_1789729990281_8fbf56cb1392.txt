   327→   327→function ApplyLeaveDialog({
   328→   328→  open, onOpenChange, balances, onApplied,
   329→   329→}: {
   330→   330→  open: boolean;
   331→   331→  onOpenChange: (v: boolean) => void;
   332→   332→  balances: LeaveBalance[];
   333→   333→  onApplied: () => void;
   334→   334→}) {
   335→   335→  const today = useMemo(() => isoDay(dayIST(new Date())), []);
   336→   336→  const [form, setForm] = useState({
   337→   337→    leaveTypeId: "", fromDate: "", toDate: "", dayMode: "FULL" as DayMode,
   338→   338→    reason: "", contact: "", substituteId: "none", attachmentName: "",
   339→   339→  });
   340→   340→  const [errors, setErrors] = useState<{ leaveTypeId?: string; dates?: string; reason?: string; balance?: string }>({});
   341→   341→
   342→   342→  const subsQuery = useQuery({
   343→   343→    queryKey: ["leave", "substitutes"],
   344→   344→    queryFn: () => apiGet<{ substitutes: Substitute[] }>("/api/leave?action=substitutes"),
   345→   345→    enabled: open,
   346→   346→  });
   347→   347→
   348→   348→  const selectedBalance = balances.find((b) => b.leaveTypeId === form.leaveTypeId) ?? null;
   349→   349→
   350→   350→  const rangeDays = useMemo(() => {
   351→   351→    if (!form.fromDate || !form.toDate) return 0;
   352→   352→    const from = new Date(`${form.fromDate}T00:00:00Z`).getTime();
   353→   353→    const to = new Date(`${form.toDate}T00:00:00Z`).getTime();
   354→   354→    if (Number.isNaN(from) || Number.isNaN(to) || from > to) return 0;
   355→   355→    return Math.round((to - from) / 86400000) + 1;
   356→   356→  }, [form.fromDate, form.toDate]);
   357→   357→  const days = rangeDays * (form.dayMode === "FULL" ? 1 : 0.5);
   358→   358→  const balanceShort = selectedBalance ? days > selectedBalance.available : false;
   359→   359→
   360→   360→  function reset() {
   361→   361→    setForm({ leaveTypeId: "", fromDate: "", toDate: "", dayMode: "FULL", reason: "", contact: "", substituteId: "none", attachmentName: "" });
   362→   362→    setErrors({});
   363→   363→  }
   364→   364→
   365→   365→  const mutation = useMutation({
   366→   366→    mutationFn: () =>
   367→   367→      apiPost<{ code: string; days: number }>("/api/leave", {
   368→   368→        action: "create",
   369→   369→        leaveTypeId: form.leaveTypeId,
   370→   370→        fromDate: form.fromDate,
   371→   371→        toDate: form.toDate || form.fromDate,
   372→   372→        dayMode: form.dayMode,
   373→   373→        reason: form.reason.trim(),
   374→   374→        contactDuringLeave: form.contact.trim() || undefined,
   375→   375→        substituteId: form.substituteId === "none" ? undefined : form.substituteId,
   376→   376→        attachmentName: form.attachmentName.trim() || undefined,
   377→   377→      }),
   378→   378→    onSuccess: (res) => {
   379→   379→      toast.success(`Leave request ${res.code} submitted`, {
   380→   380→        description: `${res.days} day(s) blocked · awaiting manager approval.`,
   381→   381→      });
   382→   382→      onOpenChange(false);
   383→   383→      reset();
   384→   384→      onApplied();
   385→   385→    },
   386→   386→    onError: (err) => {
   387→   387→      toast.error(err instanceof ApiError ? err.message : "Could not submit the request.");
   388→   388→    },
   389→   389→  });
   390→   390→
   391→   391→  function submit() {
   392→   392→    const errs: typeof errors = {};
   393→   393→    if (!form.leaveTypeId) errs.leaveTypeId = "Select a leave type.";
   394→   394→    if (!form.fromDate || !form.toDate) errs.dates = "Select both dates.";
   395→   395→    else if (form.fromDate > form.toDate) errs.dates = "From date cannot be after to date.";
   396→   396→    if (form.reason.trim().length < 5) errs.reason = "Reason must be at least 5 characters.";
   397→   397→    if (selectedBalance && days > selectedBalance.available) {
   398→   398→      errs.balance = `Only ${selectedBalance.available} day(s) available for ${selectedBalance.name}.`;
   399→   399→    }
